### Task 2 — `src/state/persistStorage.ts`

Bridges `localStorage` ↔ engine codec for the Zustand `persist` middleware.

```ts
import { createJSONStorage, type StateStorage } from 'zustand/middleware'
import { replacer, reviver } from '@/engine/codec'

const STORAGE_KEY = 'rkn-tycoon@v1' // intentionally includes the version

const wrappedLocalStorage: StateStorage = {
  getItem: (name) => {
    if (typeof window === 'undefined') return null
    return window.localStorage.getItem(name)
  },
  setItem: (name, value) => {
    if (typeof window === 'undefined') return
    window.localStorage.setItem(name, value)
  },
  removeItem: (name) => {
    if (typeof window === 'undefined') return
    window.localStorage.removeItem(name)
  },
}

export const persistStorage = createJSONStorage(
  () => wrappedLocalStorage,
  { replacer, reviver },
)

export const PERSIST_STORAGE_KEY = STORAGE_KEY

export function readRawSave(): string | null {
  if (typeof window === 'undefined') return null
  return window.localStorage.getItem(STORAGE_KEY)
}

export function storageSizeBytes(): number {
  const raw = readRawSave()
  return raw ? new Blob([raw]).size : 0
}
```

The `typeof window === 'undefined'` guards make the module SSR-safe even though it's only imported from client components — RSC import would otherwise crash on `window`.

### Task 3 — `src/state/gameStore.ts`

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { GameState } from '@/types/save'
import { initialState, CURRENT_SAVE_VERSION } from '@/types/save'
import { applyTick, applyClick } from '@/engine'
import { persistStorage, PERSIST_STORAGE_KEY } from './persistStorage'

interface GameStore extends GameState {
  // Actions
  tick: (dtMs: number) => void
  click: () => void
  /** Dev-only: hand-set blocks for testing (used by the console; remove from public bundle in Phase 7). */
  devSetBlocks: (raw: string) => void
  /** Returns true after onRehydrateStorage has run. */
  hydrated: boolean
}

export const useGameStore = create<GameStore>()(
  persist(
    (set, get) => ({
      ...initialState(),
      hydrated: false,

      tick: (dtMs) => set((s) => applyTick(s, dtMs)),
      click: () => set((s) => applyClick(s)),

      devSetBlocks: (raw) => {
        // eslint-disable-next-line @typescript-eslint/no-require-imports
        const Decimal = require('break_infinity.js').default
        set({ blocks: new Decimal(raw) })
      },
    }),
    {
      name: PERSIST_STORAGE_KEY,
      version: CURRENT_SAVE_VERSION,
      storage: persistStorage,
      // partialize: keep functions out of the persisted payload
      partialize: (s) => ({
        blocks: s.blocks,
        totalBlocksEver: s.totalBlocksEver,
        clickValue: s.clickValue,
        cps: s.cps,
        prestigeMult: s.prestigeMult,
        lastTick: s.lastTick,
        tickCount: s.tickCount,
        uptimeStartMs: s.uptimeStartMs,
      }),
      // migrate: identity for now; Phase 2+ adds branches
      migrate: (persisted, fromVersion) => {
        // Phase 1: only version 1 exists. If persisted is from an unknown version, drop it.
        if (fromVersion !== CURRENT_SAVE_VERSION) {
          console.warn(`[save] dropping save from version ${fromVersion}`)
          return initialState() as unknown as GameStore
        }
        return persisted as GameStore
      },
      onRehydrateStorage: () => (rehydrated, err) => {
        if (err) {
          console.error('[save] rehydrate failed, starting fresh:', err)
          // We cannot easily reset here because the store is currently being populated by Zustand.
          // The migrate() branch above already handles unknown versions; for JSON-parse errors the persist
          // middleware swallows and uses the initialState() values from create(). Verified in 01-05 tests.
        }
        // Mark hydration complete
        useGameStore.setState({ hydrated: true })
      },
    },
  ),
)

if (typeof window !== 'undefined') {
  ;(window as unknown as { useGameStore: typeof useGameStore }).useGameStore = useGameStore
}
```

The `window.useGameStore = useGameStore` line satisfies the Phase 1 verification requirement ("`useGameStore.getState().tick(100)` is callable from the console").

### Task 5 — `src/app/api/health/route.ts`

```ts
import { NextResponse } from 'next/server'
import { db } from '@/lib/db'

// Force dynamic — health checks shouldn't be cached
export const dynamic = 'force-dynamic'

export async function GET() {
  const startedAt = Date.now()
  try {
    await db.$queryRaw`SELECT 1`
    return NextResponse.json(
      {
        ok: true,
        version: process.env.npm_package_version ?? 'unknown',
        db: 'connected',
        latencyMs: Date.now() - startedAt,
      },
      { status: 200 },
    )
  } catch (err) {
    console.error('[health] db check failed:', err)
    return NextResponse.json(
      {
        ok: false,
        version: process.env.npm_package_version ?? 'unknown',
        db: 'disconnected',
        error: err instanceof Error ? err.message : 'unknown',
      },
      { status: 503 },
    )
  }
}
```

API-03 (JSON envelope) and API-05 (health route) both satisfied.
