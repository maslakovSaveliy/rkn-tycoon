### Task 8 — ESLint 9 flat config + 7-zone boundary

```bash
pnpm add -D eslint@^9 typescript-eslint eslint-plugin-import eslint-config-next
```

`eslint.config.js`:
```js
import nextPlugin from 'eslint-config-next'
import tseslint from 'typescript-eslint'
import importPlugin from 'eslint-plugin-import'

export default [
  ...tseslint.configs.recommendedTypeChecked,
  ...nextPlugin,
  {
    plugins: { import: importPlugin },
    languageOptions: { parserOptions: { project: './tsconfig.json' } },
    rules: {
      '@typescript-eslint/no-floating-promises': 'error',
      // Engine/state/lib must not import React/Next/UI/server
      'import/no-restricted-paths': ['error', {
        zones: [
          { target: './src/engine', from: './src/ui',       message: 'engine must stay pure (no React/UI)' },
          { target: './src/engine', from: './src/state',    message: 'engine must not depend on state' },
          { target: './src/engine', from: './src/app',      message: 'engine must not import from app/' },
          { target: './src/engine', from: './src/server',   message: 'engine must not import from server/' },
          { target: './src/engine', from: './node_modules/react', message: 'engine must not import react' },
          { target: './src/engine', from: './node_modules/next',  message: 'engine must not import next/*' },

          { target: './src/state',  from: './src/ui',       message: 'state must not import UI' },
          { target: './src/state',  from: './src/app',      message: 'state must not import app/' },
          { target: './src/state',  from: './src/server',   message: 'state must not import server/' },

          { target: './src/lib',    from: './src/ui',       message: 'lib must stay UI-agnostic' },
          { target: './src/lib',    from: './src/state',    message: 'lib must not import state' },
          { target: './src/lib',    from: './src/app',      message: 'lib must not import app/' },

          { target: './src/data',   from: './src/ui',       message: 'data must not import UI' },
          { target: './src/data',   from: './src/state',    message: 'data must not import state' },
          { target: './src/data',   from: './src/server',   message: 'data must not import server/' },
          { target: './src/data',   from: './src/app',      message: 'data must not import app/' },
        ],
      }],
      // Forbid raw DOM globals in engine/lib/data
      'no-restricted-globals': ['error', 'document', 'window'],
    },
  },
  // server/ files must declare 'server-only'
  // (enforced by convention + Prisma import being unusable client-side; lint rule deferred to Phase 9)
```

### Task 10 — Docker Compose: Postgres 16-alpine

`docker-compose.yml` in repo root:
```yaml
services:
  postgres:
    image: postgres:16-alpine
    container_name: rkn-tycoon-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: ${POSTGRES_DB:-rkn_tycoon}
      POSTGRES_USER: ${POSTGRES_USER:-rkn}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-rkn_dev_password}
    ports:
      - "5432:5432"
    volumes:
      - ./.data/postgres:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-rkn} -d ${POSTGRES_DB:-rkn_tycoon}"]
      interval: 10s
      timeout: 5s
      retries: 5
```

`.env.example`:
```
# Database
POSTGRES_DB=rkn_tycoon
POSTGRES_USER=rkn
POSTGRES_PASSWORD=rkn_dev_password
DATABASE_URL="postgresql://rkn:rkn_dev_password@localhost:5432/rkn_tycoon?schema=public"
```

`.env.local` is gitignored; user copies from `.env.example` manually.
