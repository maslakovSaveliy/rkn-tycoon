### Task 2 — `src/types/save.ts`

The shared Save shape. Versioned. Imported by engine, state, and (later) server.

```ts
import type Decimal from 'break_infinity.js'

export const CURRENT_SAVE_VERSION = 1 as const
export type SaveVersion = typeof CURRENT_SAVE_VERSION | 0 // 0 = legacy / not present

export interface GameState {
  blocks: Decimal
  totalBlocksEver: Decimal
  clickValue: Decimal
  cps: Decimal
  prestigeMult: number  // plain number; Decimal would be overkill, multiplier stays small
  lastTick: number      // Date.now() at last persisted tick (for offline progress, Phase 4)
  tickCount: number     // monotonic tick counter (for Debug HUD)
  uptimeStartMs: number // performance.timeOrigin-relative start of current session
}

export interface SavePayload {
  version: SaveVersion
  state: GameState
}

export function initialState(): GameState {
  // Decimal constructor must be called lazily — top-level evaluation would happen in engine init
  // We export a factory to make initial state explicit for tests + new-player flow.
  const Decimal = require('break_infinity.js').default // see codec.ts for the import-safe pattern
  return {
    blocks: new Decimal(0),
    totalBlocksEver: new Decimal(0),
    clickValue: new Decimal(1),
    cps: new Decimal(0),
    prestigeMult: 1.0,
    lastTick: Date.now(),
    tickCount: 0,
    uptimeStartMs: 0, // set on first tick
  }
}
```

### Task 3 — `src/engine/codec.ts` (the `__D` marker)

```ts
import Decimal from 'break_infinity.js'

const DECIMAL_MARKER = '__D' as const

interface DecimalMarker {
  readonly [DECIMAL_MARKER]: string
}

function isDecimalMarker(v: unknown): v is DecimalMarker {
  return typeof v === 'object' && v !== null && DECIMAL_MARKER in v && typeof (v as DecimalMarker)[DECIMAL_MARKER] === 'string'
}

export function replacer(_key: string, value: unknown): unknown {
  if (value instanceof Decimal) {
    return { [DECIMAL_MARKER]: value.toString() }
  }
  return value
}

export function reviver(_key: string, value: unknown): unknown {
  if (isDecimalMarker(value)) {
    return new Decimal(value[DECIMAL_MARKER])
  }
  return value
}

export function encode(state: unknown): string {
  return JSON.stringify(state, replacer)
}

export function decode<T>(raw: string): T {
  return JSON.parse(raw, reviver) as T
}
```

### Task 4 — `src/engine/migrations.ts` (stub)

```ts
import type { SavePayload, GameState, SaveVersion } from '@/types/save'
import { CURRENT_SAVE_VERSION, initialState } from '@/types/save'

/**
 * Migrate a save payload from an older version to the current shape.
 * Phase 1: identity (no real migrations exist yet). Future migrations push entries here.
 */
export function migrate(payload: { version: number; state: unknown }): SavePayload {
  let { version, state } = payload

  // Migration chain — append a new branch each time the save shape changes.
  // Example for future use:
  // if (version === 1) { state = migrateV1toV2(state); version = 2 }

  if (version !== CURRENT_SAVE_VERSION) {
    // Unknown future version OR a version we cannot reach via the chain → throw, caller decides.
    throw new MigrationError(`Cannot migrate from save version ${version} to ${CURRENT_SAVE_VERSION}`)
  }

  return { version: CURRENT_SAVE_VERSION, state: state as GameState }
}

export class MigrationError extends Error {
  constructor(msg: string) {
    super(msg)
    this.name = 'MigrationError'
  }
}

/** Safe parse: returns either the migrated payload or a fresh initial state (with reason logged). */
export function safeRehydrate(raw: string | null): SavePayload {
  if (!raw) return { version: CURRENT_SAVE_VERSION, state: initialState() }
  try {
    const parsed = JSON.parse(raw)
    // Decoding through the reviver happens at the caller (state/persistStorage.ts)
    // Here we receive an already-decoded object; we just check shape.
    if (typeof parsed !== 'object' || parsed === null || typeof parsed.version !== 'number') {
      throw new Error('save shape invalid')
    }
    return migrate(parsed as { version: number; state: unknown })
  } catch (err) {
    // Engine logs through console; UI surface (toast/modal) is Phase 2+
    console.error('[save] corrupted save, starting fresh:', err)
    return { version: CURRENT_SAVE_VERSION, state: initialState() }
  }
}
```

### Task 5 — `src/engine/economy.ts` (with prestigeMult from day 1)

CORE-06: every income formula multiplies through `prestigeMult` even though it's hardcoded `1.0` in Phase 1. Future phases (prestige, events) introduce real multipliers.

```ts
import Decimal from 'break_infinity.js'
import type { GameState } from '@/types/save'

/**
 * Apply one tick of duration `dtMs` to the state.
 * Pure function: returns a new state object, does NOT mutate input.
 *
 * Income = (clickValue applied at click time, not here) + cps * dtSec * prestigeMult
 * Phase 1: only the passive CPS path runs (no click handler yet).
 */
export function applyTick(state: GameState, dtMs: number): GameState {
  const dtSec = dtMs / 1000
  const passive = state.cps.mul(dtSec).mul(state.prestigeMult)
  const blocks = state.blocks.add(passive)
  const totalBlocksEver = state.totalBlocksEver.add(passive)
  return {
    ...state,
    blocks,
    totalBlocksEver,
    lastTick: Date.now(),
    tickCount: state.tickCount + 1,
  }
}

/** Click handler — invoked by UI on pointerdown. Phase 1: dev-only, not wired to a button. */
export function applyClick(state: GameState): GameState {
  const gain = state.clickValue.mul(state.prestigeMult)
  return {
    ...state,
    blocks: state.blocks.add(gain),
    totalBlocksEver: state.totalBlocksEver.add(gain),
  }
}

/** Convenience for tests / Debug HUD: current blocks-per-second after prestige mult. */
export function effectiveCps(state: GameState): Decimal {
  return state.cps.mul(state.prestigeMult)
}
```

### Task 6 — `src/engine/tick.ts` (rAF + accumulator)

```ts
const STEP_MS = 100  // 10 Hz logic tick

export interface TickHandle {
  stop(): void
}

/**
 * Start the rAF loop. Calls `onSteps(stepsRan)` whenever one or more 100ms steps
 * have accumulated. Returns a handle with `stop()` to cancel.
 */
export function startTickLoop(onSteps: (steps: number) => void): TickHandle {
  let acc = 0
  let last = performance.now()
  let rafId = 0
  let stopped = false

  const frame = (now: number) => {
    if (stopped) return
    acc += now - last
    last = now
    let steps = 0
    while (acc >= STEP_MS) {
      acc -= STEP_MS
      steps++
      // Safety: avoid catch-up storms if the tab was hidden for hours
      if (steps >= 600) {
        acc = 0
        break
      }
    }
    if (steps > 0) onSteps(steps)
    rafId = requestAnimationFrame(frame)
  }

  rafId = requestAnimationFrame(frame)

  return {
    stop() {
      stopped = true
      cancelAnimationFrame(rafId)
    },
  }
}

/**
 * Pure function for testing the accumulator math without rAF.
 * Returns (steps ran, remainder ms).
 */
export function runAccumulator(deltaMs: number, accumulatorMs: number, stepMs: number = STEP_MS): { steps: number; remainder: number } {
  let acc = accumulatorMs + deltaMs
  let steps = 0
  while (acc >= stepMs) {
    acc -= stepMs
    steps++
    if (steps >= 600) { acc = 0; break }
  }
  return { steps, remainder: acc }
}

export const TICK_STEP_MS = STEP_MS
```

**Why `requestAnimationFrame` is OK to reference here despite the ESLint boundary**: `requestAnimationFrame` is a DOM global. We need it. The boundary rule for `engine/` forbids the `react`, `next`, `motion`, `howler` etc. — not browser intrinsics. We add an explicit `eslint-disable-next-line no-restricted-globals` if the global rule covers `requestAnimationFrame` (it doesn't by default — `no-restricted-globals` was scoped to `document`, `window` in 01-01). Engine tests in 01-05 will stub `requestAnimationFrame` for jsdom or fake timers.
