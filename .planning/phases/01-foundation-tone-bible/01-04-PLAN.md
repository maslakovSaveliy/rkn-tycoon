## Tasks

### Task 1 — `src/ui/AppShell.tsx` (top-level client component)

```tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { useGameStore } from '@/state/gameStore'
import { startTickLoop, TICK_STEP_MS, type TickHandle } from '@/engine'
import { DebugHud } from './DebugHud'

export function AppShell() {
  const hydrated = useGameStore((s) => s.hydrated)
  const tick = useGameStore((s) => s.tick)
  const handleRef = useRef<TickHandle | null>(null)
  const [fps, setFps] = useState(0)
  const fpsState = useRef({ frames: 0, last: 0 })

  useEffect(() => {
    if (!hydrated) return // wait for persist to finish
    handleRef.current = startTickLoop((steps) => {
      tick(steps * TICK_STEP_MS)
      // FPS proxy: count rAF frames over a 1s window
      fpsState.current.frames++
      const now = performance.now()
      if (now - fpsState.current.last >= 1000) {
        setFps(fpsState.current.frames)
        fpsState.current.frames = 0
        fpsState.current.last = now
      }
    })
    return () => handleRef.current?.stop()
  }, [hydrated, tick])

  if (!hydrated) {
    return <main className="min-h-screen flex items-center justify-center">загрузка сейва…</main>
  }

  return (
    <main className="min-h-screen p-6 flex flex-col gap-6">
      <header>
        <h1 className="text-2xl">RKN Tycoon — игровой режим</h1>
        <p className="text-sm opacity-60">Phase 1 placeholder. Игровая кнопка появится в Phase 2.</p>
      </header>
      <DebugHud fps={fps} />
    </main>
  )
}
```

The store's `hydrated` flag (set by `onRehydrateStorage` in 01-03) gates the tick loop — we don't want to start ticking before the persisted state has been read, otherwise the first tick would briefly run against `initialState()` and overwrite it.

### Task 2 — `src/ui/DebugHud.tsx`

```tsx
'use client'

import { useEffect, useState } from 'react'
import { useGameStore } from '@/state/gameStore'
import { effectiveCps } from '@/engine'
import { storageSizeBytes } from '@/state/persistStorage'

interface Props {
  fps: number
}

export function DebugHud({ fps }: Props) {
  const blocks = useGameStore((s) => s.blocks)
  const tickCount = useGameStore((s) => s.tickCount)
  const state = useGameStore.getState
  const [uptimeMs, setUptimeMs] = useState(0)
  const [storage, setStorage] = useState(0)
  const startedAt = useState(() => Date.now())[0]

  useEffect(() => {
    const id = setInterval(() => {
      setUptimeMs(Date.now() - startedAt)
      setStorage(storageSizeBytes())
    }, 250)
    return () => clearInterval(id)
  }, [startedAt])

  const cps = effectiveCps(state())

  return (
    <pre
      className="font-mono text-sm leading-tight border border-current p-3 inline-block"
      role="status"
      aria-live="polite"
    >
      blocks:  {blocks.toString()}
      {'\n'}BPS:     {cps.toString()}
      {'\n'}tick:    #{tickCount}
      {'\n'}FPS:     {fps}
      {'\n'}save v:  1
      {'\n'}storage: {(storage / 1024).toFixed(2)} KB
      {'\n'}uptime:  {formatUptime(uptimeMs)}
    </pre>
  )
}

function formatUptime(ms: number): string {
  const sec = Math.floor(ms / 1000)
  const h = Math.floor(sec / 3600)
  const m = Math.floor((sec % 3600) / 60)
  const s = sec % 60
  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
}
```

The `pre` keeps the alignment without ASCII-box-drawing chars (Phase 7 will replace with full ASCII frame). The polling `setInterval(250)` only refreshes uptime + storage size — actual game state comes through Zustand selectors that re-render on store change.

**Note**: the live `blocks` here renders `Decimal.toString()` which can produce `1.23e+456` for huge numbers — that's intentional in Phase 1. Phase 2 adds the К/М/Б/Т formatter.

### Task 7 — Visibility-change autosave hook (lightweight)

Per CLAUDE.md §4: persist already saves on every `set`. We add a `visibilitychange` flush to guarantee the save is current before a tab unload — this is the Phase 1 version of SAVE-01 (the 10s timer comes in Phase 3).

```tsx
// inside src/ui/AppShell.tsx, augment the existing useEffect
useEffect(() => {
  const flush = () => {
    // Zustand persist's storage.setItem already runs on every set();
    // calling getState() forces the persist middleware to commit any pending state.
    useGameStore.persist?.rehydrate?.() // no-op safety
    // For an explicit flush, we just touch the store — persist serializes synchronously.
    useGameStore.setState({})
  }
  document.addEventListener('visibilitychange', flush)
  window.addEventListener('beforeunload', flush)
  return () => {
    document.removeEventListener('visibilitychange', flush)
    window.removeEventListener('beforeunload', flush)
  }
}, [])
```

(Zustand `persist` actually writes on every `setState`, so the explicit flush is overkill — but the visibility listener is the Phase-1 affordance toward the proper SAVE-01 implementation in Phase 3. Keep it; the cost is zero.)
