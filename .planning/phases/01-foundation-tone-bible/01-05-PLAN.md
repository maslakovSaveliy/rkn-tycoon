### Task 3 — Test 1: codec round-trip (single Decimal)

`src/engine/codec.test.ts`:
```ts
import { describe, it, expect } from 'vitest'
import Decimal from 'break_infinity.js'
import { encode, decode } from './codec'

describe('codec', () => {
  it('round-trips a single Decimal value through JSON', () => {
    const original = new Decimal('1.23e456')
    const wrapper = { x: original }
    const json = encode(wrapper)
    expect(json).toContain('__D')
    const back = decode<{ x: Decimal }>(json)
    expect(back.x).toBeInstanceOf(Decimal)
    expect(back.x.eq(original)).toBe(true)
  })

  it('handles Decimal that equals zero', () => {
    const original = new Decimal(0)
    const back = decode<{ x: Decimal }>(encode({ x: original }))
    expect(back.x.eq(0)).toBe(true)
  })
})
```

### Task 4 — Test 2: codec round-trip nested

Append to `src/engine/codec.test.ts`:
```ts
describe('codec — nested', () => {
  it('round-trips Decimals inside objects and arrays', () => {
    const value = {
      cps: new Decimal('5.5e10'),
      upgrades: [new Decimal('1'), new Decimal('2e3'), new Decimal('0')],
      nested: { deep: { d: new Decimal('1e1000') } },
      plain: 42,
    }
    const back = decode<typeof value>(encode(value))
    expect(back.cps.eq('5.5e10')).toBe(true)
    expect(back.upgrades[0].eq(1)).toBe(true)
    expect(back.upgrades[1].eq('2e3')).toBe(true)
    expect(back.upgrades[2].eq(0)).toBe(true)
    expect(back.nested.deep.d.eq('1e1000')).toBe(true)
    expect(back.plain).toBe(42)
  })

  it('does not corrupt plain objects that happen to have an __D key', () => {
    // Edge case: a future migration might add a literal field named __D for some reason.
    // The reviver should still hydrate it as a Decimal — that's the contract.
    // If we ever need to bypass the reviver, we'll change the marker.
    const json = JSON.stringify({ x: { __D: '7' } })
    const back = decode<{ x: Decimal }>(json)
    expect(back.x).toBeInstanceOf(Decimal)
    expect(back.x.eq(7)).toBe(true)
  })
})
```

### Task 5 — Test 3: tick accumulator math

`src/engine/tick.test.ts`:
```ts
import { describe, it, expect } from 'vitest'
import { runAccumulator } from './tick'

describe('runAccumulator', () => {
  it('250ms delta with 0ms initial gives 2 steps + 50ms remainder', () => {
    const r = runAccumulator(250, 0, 100)
    expect(r.steps).toBe(2)
    expect(r.remainder).toBe(50)
  })

  it('99ms delta gives 0 steps and full delta as remainder', () => {
    expect(runAccumulator(99, 0, 100)).toEqual({ steps: 0, remainder: 99 })
  })

  it('catch-up storms cap at 600 steps to avoid lockup after long tab-hide', () => {
    // 1 hour = 36_000 steps if we let it run; the safety cap should snap to 600.
    const r = runAccumulator(3_600_000, 0, 100)
    expect(r.steps).toBe(600)
    expect(r.remainder).toBe(0)
  })

  it('accumulator carries remainder across calls', () => {
    const a = runAccumulator(150, 0, 100)  // 1 step, 50 remainder
    const b = runAccumulator(75, a.remainder, 100)  // 1 step (125 >= 100), 25 remainder
    expect(a).toEqual({ steps: 1, remainder: 50 })
    expect(b).toEqual({ steps: 1, remainder: 25 })
  })
})
```

### Task 6 — Test 4: migrate identity

`src/engine/migrations.test.ts`:
```ts
import { describe, it, expect } from 'vitest'
import Decimal from 'break_infinity.js'
import { migrate, MigrationError, safeRehydrate } from './migrations'
import { initialState, CURRENT_SAVE_VERSION } from '@/types/save'

describe('migrate', () => {
  it('passes a current-version payload through unchanged (identity)', () => {
    const state = { ...initialState(), blocks: new Decimal('42') }
    const result = migrate({ version: CURRENT_SAVE_VERSION, state })
    expect(result.version).toBe(CURRENT_SAVE_VERSION)
    expect(result.state).toBe(state) // same reference — true identity
  })

  it('throws MigrationError on an unknown version', () => {
    expect(() => migrate({ version: 999, state: {} })).toThrow(MigrationError)
  })
})

describe('safeRehydrate', () => {
  it('returns a fresh initial state when raw is null', () => {
    const r = safeRehydrate(null)
    expect(r.version).toBe(CURRENT_SAVE_VERSION)
    expect(r.state.blocks.eq(0)).toBe(true)
  })
})
```

### Task 7 — Test 5: corrupted save fallback

Append to `src/engine/migrations.test.ts`:
```ts
import { vi } from 'vitest'

describe('safeRehydrate — corrupted input', () => {
  it('falls back to initial state and logs an error on invalid JSON', () => {
    const errSpy = vi.spyOn(console, 'error').mockImplementation(() => {})
    const r = safeRehydrate('this is not json {')
    expect(r.version).toBe(CURRENT_SAVE_VERSION)
    expect(r.state.blocks.eq(0)).toBe(true)
    expect(errSpy).toHaveBeenCalledOnce()
    errSpy.mockRestore()
  })

  it('falls back when JSON is valid but shape is wrong', () => {
    const errSpy = vi.spyOn(console, 'error').mockImplementation(() => {})
    const r = safeRehydrate('{"random":"object"}')
    expect(r.version).toBe(CURRENT_SAVE_VERSION)
    expect(r.state.blocks.eq(0)).toBe(true)
    errSpy.mockRestore()
  })
})
```
